python -m venv venv

venv\Scripts\activate

#these are the CPU versions of torch
#for a PC with a Nvidia GPU we'll use a different Version with CUDA that should be faster
pip uninstall torch torchvision torchaudio -y 
pip install torch==2.1.2 torchvision==0.16.2 torchaudio==2.1.2 --index-url https://download.pytorch.org/whl/cpu

pip install -r requirements.txt


#we need these exact versions else there are requirement conflicts
pip uninstall transformers -y
pip install transformers==4.38.2

pip uninstall gradio jinja2 fastapi starlette uvicorn -y
pip install gradio==4.44.1
pip install jinja2==3.1.4
pip install fastapi==0.110.0
pip install starlette==0.36.3
pip install uvicorn==0.29.0

pip uninstall gradio -y
